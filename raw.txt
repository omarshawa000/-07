    while LoadGui.Parent do
        if LoadGui.Enabled then
            DotsText.Text = "." task.wait(0.3)
            DotsText.Text = ".." task.wait(0.3)
            DotsText.Text = "..." task.wait(0.3)
        else task.wait(0.1) end
    end
end)

local function startLoading(lang)
    isEnglish = (lang == "en")
    TweenService:Create(LangFrame, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.In), {Size = UDim2.new(0, 0, 0, 0)}):Play()
    task.wait(0.4)
    LangGui:Destroy()
    
    LoadGui.Enabled = true
    LoadText.Text = T("انتظر قليلا", "Please wait a moment")
    task.wait(5)
    
    TweenService:Create(LoadShade, TweenInfo.new(0.5), {BackgroundTransparency = 1}):Play()
    TweenService:Create(LoadText, TweenInfo.new(0.5), {TextTransparency = 1}):Play()
    TweenService:Create(DotsText, TweenInfo.new(0.5), {TextTransparency = 1}):Play()
    task.wait(0.5)
    LoadGui:Destroy()
    createMainUI()
end

ArBtn.MouseButton1Click:Connect(function() startLoading("ar") end)
EnBtn.MouseButton1Click:Connect(function() startLoading("en") end)

-- 3. الواجهة الرئيسية (تصميم خرافي)
function createMainUI()
    local ScreenGui = Instance.new("ScreenGui")
    ScreenGui.Name = "UltimateObbyGod"
    ScreenGui.ResetOnSpawn = false
    ScreenGui.Parent = LocalPlayer:WaitForChild("PlayerGui")

    local WelcomeFrame = Instance.new("Frame")
    WelcomeFrame.Size = UDim2.new(0, 300, 0, 80)
    WelcomeFrame.Position = UDim2.new(1, 20, 1, -100)
    WelcomeFrame.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
    WelcomeFrame.BackgroundTransparency = 0.1
    WelcomeFrame.BorderSizePixel = 2
    WelcomeFrame.BorderColor3 = Color3.fromRGB(0, 255, 0)
    WelcomeFrame.Parent = ScreenGui

    local WelcomeCorner = Instance.new("UICorner")
    WelcomeCorner.CornerRadius = UDim.new(0, 10)
    WelcomeCorner.Parent = WelcomeFrame

    local WelcomeTitle = Instance.new("TextLabel")
    WelcomeTitle.Size = UDim2.new(1, 0, 0, 35)
    WelcomeTitle.Position = UDim2.new(0, 0, 0, 5)
    WelcomeTitle.BackgroundTransparency = 1
    WelcomeTitle.Text = T("مرحبا بك💚", "Welcome💚")
    WelcomeTitle.TextColor3 = Color3.fromRGB(0, 255, 0)
    WelcomeTitle.TextSize = 20
    WelcomeTitle.Font = Enum.Font.FredokaOne
    WelcomeTitle.Parent = WelcomeFrame

    local WelcomeSub = Instance.new("TextLabel")
    WelcomeSub.Size = UDim2.new(1, -10, 0, 30)
    WelcomeSub.Position = UDim2.new(0, 5, 0, 40)
    WelcomeSub.BackgroundTransparency = 1
    WelcomeSub.Text = T("السكربت يمنعك من الموت، استمتع", "The script prevents death, enjoy")
    WelcomeSub.TextColor3 = Color3.fromRGB(200, 200, 200)
    WelcomeSub.TextSize = 14
    WelcomeSub.Font = Enum.Font.FredokaOne
    WelcomeSub.TextWrapped = true
    WelcomeSub.Parent = WelcomeFrame

    TweenService:Create(WelcomeFrame, TweenInfo.new(0.8, Enum.EasingStyle.Back), {Position = UDim2.new(1, -320, 1, -100)}):Play()
    task.wait(4)
    TweenService:Create(WelcomeFrame, TweenInfo.new(0.8, Enum.EasingStyle.Back), {Position = UDim2.new(1, 20, 1, -100)}):Play()
    task.delay(0.8, function() WelcomeFrame:Destroy() end)

    -- Main Frame
    local MainFrame = Instance.new("Frame")
    MainFrame.Size = UDim2.new(0, 260, 0, 160)
    MainFrame.Position = UDim2.new(0.5, -130, 0.5, -80)
    MainFrame.BackgroundColor3 = Color3.fromRGB(5, 5, 5)
    MainFrame.BackgroundTransparency = 0.1
    MainFrame.BorderSizePixel = 0
    MainFrame.Active = true
    MainFrame.Draggable = true
    MainFrame.Parent = ScreenGui

    local MainCorner = Instance.new("UICorner")
    MainCorner.CornerRadius = UDim.new(0, 15)
    MainCorner.Parent = MainFrame

    local MainStroke = Instance.new("UIStroke")
    MainStroke.Color = Color3.fromRGB(0, 255, 0)
    MainStroke.Thickness = 3
    MainStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    MainStroke.Parent = MainFrame

    local Title = Instance.new("TextLabel")
    Title.Size = UDim2.new(1, 0, 0, 40)

    Title.Text = T("🛡️ مضاد للموت", "🛡️ Anti-Death")

    Title.TextSize = 22
    Title.Font = Enum.Font.FredokaOne
    Title.Parent = MainFrame

    local Btn = Instance.new("TextButton")
    Btn.Size = UDim2.new(1, -40, 0, 50)

    Btn.BorderSizePixel = 2
    Btn.BorderColor3 = Color3.fromRGB(0, 255, 0)
    Btn.Text = T("✅ تشغيل", "✅ Enable")
    Btn.TextColor3 = Color3.fromRGB(0, 255, 0)
    Btn.TextSize = 18
    Btn.Font = Enum.Font.FredokaOne
    Btn.Parent = MainFrame

    local BtnCorner = Instance.new("UICorner")
    BtnCorner.CornerRadius = UDim.new(0, 10)
    BtnCorner.Parent = Btn

    local Status = Instance.new("TextLabel")
    Status.Size = UDim2.new(1, -20, 0, 25)
    Status.Position = UDim2.new(0, 10, 1, -35)
    Status.BackgroundTransparency = 1
    Status.Text = T("الحالة: معطل", "Status: Disabled")
    Status.TextColor3 = Color3.fromRGB(255, 0, 0)
    Status.TextSize = 14
    Status.Font = Enum.Font.FredokaOne
    Status.TextXAlignment = Enum.TextXAlignment.Center
    Status.Parent = MainFrame

    local function clearConns()
        for _, c in ipairs(conns) do
            pcall(function() c:Disconnect() end)
        end
        conns = {}
    end

    local function protectCharacter(char)
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum then return end

        savedMax = hum.MaxHealth
        hum.MaxHealth = math.huge
        hum.Health = math.huge
        hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
        hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
        hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
        hum.BreakJointsOnDeath = false

        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then part.CanTouch = false end
        end

        table.insert(conns, char.DescendantAdded:Connect(function(obj)
            if active and obj:IsA("BasePart") then obj.CanTouch = false end
        end))
        table.insert(conns, hum.HealthChanged:Connect(function(h)
            if active and h < math.huge then hum.Health = math.huge end
        end))
        table.insert(conns, hum.Died:Connect(function()
            if active then
                task.wait(0.01)
                pcall(function() LocalPlayer:LoadCharacter() end)
            end
        end))
    end

    local function applyGod()
        clearConns()
        local char = LocalPlayer.Character
        if char then protectCharacter(char) end
        table.insert(conns, LocalPlayer.CharacterAdded:Connect(function(newChar)
            wait(0.1)
            if active then protectCharacter(newChar) end
        end))
        table.insert(conns, RunService.Heartbeat:Connect(function()
            if not active then return end
            local c = LocalPlayer.Character
            if not c then return end
            local h = c:FindFirstChildOfClass("Humanoid")
            if h then
                h.Health = math.huge
                h.MaxHealth = math.huge
                h:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
                h:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
                h:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
                h.BreakJointsOnDeath = false
            end
            for _, part in ipairs(c:GetDescendants()) do
                if part:IsA("BasePart") then part.CanTouch = false end
            end
        end))
    end

    local function removeGod()
        clearConns()
        local char = LocalPlayer.Character
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then
                hum.MaxHealth = savedMax > 0 and savedMax or 100
                hum.Health = hum.MaxHealth
                hum:SetStateEnabled(Enum.HumanoidStateType.Dead, true)
                hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, true)
                hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true)
                hum.BreakJointsOnDeath = true
            end
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") then part.CanTouch = true end
            end
        end
    end

    Btn.MouseButton1Click:Connect(function()
        active = not active
        if active then
            applyGod()
            Btn.Text = T("❌ إيقاف", "❌ Disable")
            Btn.BorderColor3 = Color3.fromRGB(255, 0, 0)
            Btn.TextColor3 = Color3.fromRGB(255, 0, 0)
            Status.Text = T("الحالة: مفعل", "Status: Enabled")
            Status.TextColor3 = Color3.fromRGB(0, 255, 0)
            TweenService:Create(MainStroke, TweenInfo.new(0.3), {Color = Color3.fromRGB(255, 0, 0)}):Play()
        else
            removeGod()
            Btn.Text = T("✅ تشغيل", "✅ Enable")
            Btn.BorderColor3 = Color3.fromRGB(0, 255, 0)
            Btn.TextColor3 = Color3.fromRGB(0, 255, 0)
            Status.Text = T("الحالة: معطل", "Status: Disabled")
            Status.TextColor3 = Color3.fromRGB(255, 0, 0)
            TweenService:Create(MainStroke, TweenInfo.new(0.3), {Color = Color3.fromRGB(0, 255, 0)}):Play()
        end
    end)
end
